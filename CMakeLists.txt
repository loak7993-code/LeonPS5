cmake_minimum_required(VERSION 3.20)
project(LeonPS5 CXX)

set(LEONPS5_TEST_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")

function(add_test_executable target)
    add_executable(${target} ${ARGN})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${LEONPS5_TEST_OUTPUT_DIRECTORY}>")
endfunction()

option(BUILD_TESTING "Build and register relinker tests" OFF)
option(LEONPS5_RELINKER_ONLY "Build the relinker without system libraries or third-party dependencies" OFF)
option(LEONPS5_CCACHE "Cache compiler invocations with ccache" ON)
option(LEONPS5_LTO "Link-time optimization in single-config release builds" ON)
if(BUILD_TESTING)
    enable_testing()
    if(MINGW)
        get_filename_component(testCompilerDirectory "${CMAKE_CXX_COMPILER}" DIRECTORY)
        foreach(runtimeName IN ITEMS libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
            if(NOT EXISTS "${testCompilerDirectory}/${runtimeName}")
                message(FATAL_ERROR "Cannot locate MinGW runtime: ${runtimeName}")
            endif()
            file(COPY "${testCompilerDirectory}/${runtimeName}" DESTINATION "${LEONPS5_TEST_OUTPUT_DIRECTORY}")
        endforeach()
    endif()
endif()

option(LEONPS5_ENABLE_SPIRV_TOOLS "Enable SPIRV-Tools validation and optimization" OFF)
option(APS5_ENABLE_TIMING_LOG "Write frame timings asynchronously to frame-timing.log" OFF)
add_compile_definitions(APS5_ENABLE_TIMING_LOG=$<BOOL:${APS5_ENABLE_TIMING_LOG}>)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(LEONPS5_CCACHE AND NOT CMAKE_C_COMPILER_LAUNCHER AND NOT CMAKE_CXX_COMPILER_LAUNCHER)
    find_program(LEONPS5_CCACHE_PROGRAM ccache)
    if(LEONPS5_CCACHE_PROGRAM)
        set(CMAKE_C_COMPILER_LAUNCHER "${LEONPS5_CCACHE_PROGRAM}")
        set(CMAKE_CXX_COMPILER_LAUNCHER "${LEONPS5_CCACHE_PROGRAM}")
    endif()
endif()

if(LEONPS5_LTO AND CMAKE_BUILD_TYPE STREQUAL "Release" AND NOT MINGW AND NOT LEONPS5_RELINKER_ONLY)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT LEONPS5_IPO_SUPPORTED OUTPUT LEONPS5_IPO_OUTPUT)
    if(LEONPS5_IPO_SUPPORTED)
        set(CMAKE_C_COMPILE_OPTIONS_IPO "-flto=auto" "-ffat-lto-objects")
        set(CMAKE_CXX_COMPILE_OPTIONS_IPO "-flto=auto" "-ffat-lto-objects")
        set(CMAKE_C_LINK_OPTIONS_IPO "-flto=auto" "-ffat-lto-objects")
        set(CMAKE_CXX_LINK_OPTIONS_IPO "-flto=auto" "-ffat-lto-objects")
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION FALSE)
        message(STATUS "LeonPS5: link-time optimization is applied to the host tools (relinker, runner, nid_patcher), not the guest libraries")
    else()
        message(STATUS "LeonPS5: link-time optimization unavailable: ${LEONPS5_IPO_OUTPUT}")
    endif()
endif()
